function showTab(tabName) { document.querySelectorAll('.tab-content').forEach(tab => tab.style.display = 'none'); document.querySelectorAll('.nav-btn').forEach(btn => btn.classList.remove('active')); if (tabName === 'bill') { $('billTab').style.display = 'block'; document.querySelector('.nav-btn:nth-child(1)').classList.add('active'); } else if (tabName === 'items') { $('itemsTab').style.display = 'block'; document.querySelector('.nav-btn:nth-child(2)').classList.add('active'); populateDropdowns(); refreshCustomManager(); } else if (tabName === 'sales') { $('salesTab').style.display = 'block'; document.querySelector('.nav-btn:nth-child(3)').classList.add('active'); renderSalesTable(); } }
